# Technical SEO audit crawler (main.py).
#

# degrades with a printed message rather than crashing, so only the core

# SEO Automation Suite consumes.

beautifulsoup4

# falls back to writing .rtf.
python-docx

# --- Optional ---------------------------------------------------------------
# --render (headless-Chromium rendering for JavaScript-heavy pages). After
# installing the package, also run:
